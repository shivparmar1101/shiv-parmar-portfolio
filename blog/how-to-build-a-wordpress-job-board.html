1. Registering the Custom Post Type & Taxonomies

To start, we need a dedicated content structure separate from standard posts and pages. We will create a job_listing Custom Post Type (CPT) along with two custom taxonomies: Job Type (e.g., Full-Time, Remote, Contract) and Job Category (e.g., Engineering, Marketing).

Add the following code to a custom functionality plugin or your child theme's functions.php file:

function sp_register_job_board_cpt() {
    $labels = array(
        'name'               => 'Job Listings',
        'singular_name'      => 'Job Listing',
        'add_new'            => 'Add New Job',
        'add_new_item'       => 'Add New Job Listing',
        'edit_item'          => 'Edit Job Listing',
        'all_items'          => 'All Jobs',
        'menu_name'          => 'Job Board',
    );

    $args = array(
        'labels'             => $labels,
        'public'             => true,
        'has_archive'        => true,
        'rewrite'            => array('slug' => 'jobs'),
        'menu_icon'          => 'dashicons-briefcase',
        'supports'           => array('title', 'editor', 'thumbnail', 'revisions'),
        'show_in_rest'       => true, // Enables Gutenberg and REST API access
    );

    register_post_type('job_listing', $args);

    // Register Job Category Taxonomy
    register_taxonomy('job_category', 'job_listing', array(
        'label'        => 'Job Categories',
        'rewrite'      => array('slug' => 'job-category'),
        'hierarchical' => true,
        'show_in_rest' => true,
    ));

    // Register Job Type Taxonomy
    register_taxonomy('job_type', 'job_listing', array(
        'label'        => 'Job Types',
        'rewrite'      => array('slug' => 'job-type'),
        'hierarchical' => false,
        'show_in_rest' => true,
    ));
}
add_action('init', 'sp_register_job_board_cpt');

2. Adding Custom Meta Fields for Job Details

A job listing isn't complete without metadata like company name, location, salary range, and application URL. While plugins like ACF (Advanced Custom Fields) are great, implementing these natively via WordPress Meta Boxes keeps your site lightning fast.

// Register Meta Box
function sp_add_job_meta_boxes() {
    add_meta_box(
        'sp_job_details',
        'Job Details',
        'sp_render_job_meta_box',
        'job_listing',
        'normal',
        'high'
    );
}
add_action('add_meta_boxes', 'sp_add_job_meta_boxes');

// Render Meta Box Form Fields
function sp_render_job_meta_box($post) {
    wp_nonce_field('sp_save_job_meta', 'sp_job_meta_nonce');
    
    $company  = get_post_meta($post->ID, '_sp_company_name', true);
    $location = get_post_meta($post->ID, '_sp_job_location', true);
    $salary   = get_post_meta($post->ID, '_sp_job_salary', true);
    
    echo '<p><label for="sp_company">Company Name:</label><br>';
    echo '<input type="text" id="sp_company" name="sp_company_name" value="' . esc_attr($company) . '" class="widefat"></p>';

    echo '<p><label for="sp_location">Location (e.g., Remote, Rajkot, London):</label><br>';
    echo '<input type="text" id="sp_location" name="sp_job_location" value="' . esc_attr($location) . '" class="widefat"></p>';

    echo '<p><label for="sp_salary">Salary Range:</label><br>';
    echo '<input type="text" id="sp_salary" name="sp_job_salary" value="' . esc_attr($salary) . '" class="widefat"></p>';
}

// Save Meta Box Data Securely
function sp_save_job_meta_data($post_id) {
    if (!isset($_POST['sp_job_meta_nonce']) || !wp_verify_nonce($_POST['sp_job_meta_nonce'], 'sp_save_job_meta')) {
        return;
    }
    if (defined('DOING_AUTOSAVE') && DOING_AUTOSAVE) return;
    if (!current_user_can('edit_post', $post_id)) return;

    if (isset($_POST['sp_company_name'])) {
        update_post_meta($post_id, '_sp_company_name', sanitize_text_field($_POST['sp_company_name']));
    }
    if (isset($_POST['sp_job_location'])) {
        update_post_meta($post_id, '_sp_job_location', sanitize_text_field($_POST['sp_job_location']));
    }
    if (isset($_POST['sp_job_salary'])) {
        update_post_meta($post_id, '_sp_job_salary', sanitize_text_field($_POST['sp_job_salary']));
    }
}
add_action('save_post', 'sp_save_job_meta_data');

3. Building the Front-End Job Listing View

Now that our data model is established, we need a clean layout to render job listings on the front end. Create a template file named archive-job_listing.php in your theme folder.

<?php get_header(); ?>

<main class="job-board-container">
    <h1 class="page-title">Latest Job Openings</h1>

    <div class="job-listings-grid">
        <?php if (have_posts()) : while (have_posts()) : the_post(); 
            $company  = get_post_meta(get_the_ID(), '_sp_company_name', true);
            $location = get_post_meta(get_the_ID(), '_sp_job_location', true);
            $salary   = get_post_meta(get_the_ID(), '_sp_job_salary', true);
            $job_types = get_the_terms(get_the_ID(), 'job_type');
        ?>
            <article id="post-<?php the_ID(); ?>" <?php post_class('job-card'); ?>>
                <div class="job-card-header">
                    <h2 class="job-title"><a href="<?php the_permalink(); ?>"><?php the_title(); ?></a></h2>
                    <span class="company-name"><?php echo esc_html($company); ?></span>
                </div>
                
                <div class="job-card-meta">
                    <span class="location">📍 <?php echo esc_html($location); ?></span>
                    <span class="salary">💰 <?php echo esc_html($salary); ?></span>
                    <?php if ($job_types && !is_wp_error($job_types)) : ?>
                        <span class="job-badge"><?php echo esc_html($job_types[0]->name); ?></span>
                    <?php endif; ?>
                </div>

                <a href="<?php the_permalink(); ?>" class="apply-btn">View Details</a>
            </article>
        <?php endwhile; wp_reset_postdata(); else : ?>
            <p>No job listings found.</p>
        <?php endif; ?>
    </div>
</main>

<?php get_footer(); ?>
